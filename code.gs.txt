const CHANNEL_ID = 'UCxCBjGfJL25wnIgsOG8KUCw'; // ช่อง Sornorpoom Learning Youtube

/**
 * 2. ฟังก์ชัน Sync ข้อมูลจาก YouTube API ลง Google Sheets
 * (ควรตั้งเวลา Time-driven Trigger ให้รันวันละ 1 ครั้ง)
 *
 * [แก้ไข] ป้องกันการเขียนทับค่าที่แก้มือใน Sheet:
 *   - คอลัมน์ E (Category_ID) : ถ้าเคยกรอกไว้ จะเก็บค่าเดิมไว้
 *   - คอลัมน์ F (Target_Audience) : ถ้าเคยกรอกไว้ จะเก็บค่าเดิมไว้
 *   - คอลัมน์ G (Status) : ถ้าเคยเปลี่ยนเป็น Hide จะเก็บค่าเดิมไว้ (ไม่รีเซ็ตกลับเป็น Active)
 */
function syncYouTubeDataCustom() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const playlistSheet = ss.getSheetByName('PLAYLISTS');
  const videoSheet = ss.getSheetByName('VIDEOS');

  if (!playlistSheet || !videoSheet) {
    Logger.log("⚠️ กรุณาตรวจสอบว่ามี Sheet ชื่อ 'PLAYLISTS' และ 'VIDEOS'");
    return;
  }

  // ─── Step 1: Backup ค่าที่แก้มือใน PLAYLISTS (E, F, G) ก่อน Sync ─────────
  const manualPlaylistValues = {};
  if (playlistSheet.getLastRow() > 1) {
    const existingData = playlistSheet
      .getRange(2, 1, playlistSheet.getLastRow() - 1, 7)
      .getValues();
    existingData.forEach(row => {
      const plId = row[0];
      if (plId) {
        manualPlaylistValues[plId] = {
          category : row[4] ? row[4].toString().trim() : '',  // E: Category_ID
          target   : row[5] ? row[5].toString().trim() : '',  // F: Target_Audience
          status   : row[6] ? row[6].toString().trim() : ''   // G: Status
        };
      }
    });
  }
  Logger.log(`📦 Backup ข้อมูลเดิมใน PLAYLISTS: ${Object.keys(manualPlaylistValues).length} รายการ`);

  // ─── Step 2: ดึง Playlists ทั้งหมดจาก YouTube API ────────────────────────
  let playlists = [];
  let nextPageToken = '';

  do {
    let response = YouTube.Playlists.list('snippet,status', {
      channelId: CHANNEL_ID,
      maxResults: 50,
      pageToken: nextPageToken
    });
    if (response.items) playlists = playlists.concat(response.items);
    nextPageToken = response.nextPageToken;
  } while (nextPageToken);

  let playlistRows = [];
  let videoRows = [];
  let existingVideoKeys = getExistingVideoKeys(videoSheet);

  // ─── Step 3: วนลูปสร้าง playlistRows และดึง Videos ──────────────────────
  playlists.forEach(pl => {
    const plId    = pl.id;
    const plTitle = pl.snippet.title;
    const plDesc  = (pl.snippet.description || '').replace(/\n/g, ' ');
    const plThumb = pl.snippet.thumbnails && pl.snippet.thumbnails.high
                    ? pl.snippet.thumbnails.high.url : '';

    // ── ดึงค่า manual ที่เคยบันทึกไว้ (ถ้ามี) ──────────────────────────────
    const prev = manualPlaylistValues[plId] || {};

    // กฎการ Merge:
    //   E (category) → ใช้ค่าเดิมถ้ามี, ไม่งั้นใส่ว่าง (รอกรอกมือ)
    //   F (target)   → ใช้ค่าเดิมถ้ามี, ไม่งั้นใส่ว่าง
    //   G (status)   → ใช้ค่าเดิมถ้ามี, ไม่งั้นตั้งเป็น 'Active'
    const mergedCategory = prev.category || '';
    const mergedTarget   = prev.target   || '';
    const mergedStatus   = prev.status   || 'Active';

    // โครงสร้าง A-G ของ PLAYLISTS
    playlistRows.push([
      plId,           // A: Playlist_ID
      plTitle,        // B: Playlist_Title
      plDesc,         // C: Description
      plThumb,        // D: Thumbnail_URL
      mergedCategory, // E: Category_ID   ← คงค่าเดิมที่กรอกมือ
      mergedTarget,   // F: Target_Audience ← คงค่าเดิมที่กรอกมือ
      mergedStatus    // G: Status          ← คงค่าเดิม (ไม่รีเซ็ตเป็น Active)
    ]);

    // ── ดึงวิดีโอใน Playlist นี้ ─────────────────────────────────────────────
    let vNextToken = '';
    do {
      let vResponse = YouTube.PlaylistItems.list('snippet', {
        playlistId: plId,
        maxResults: 50,
        pageToken: vNextToken
      });

      if (vResponse.items) {
        vResponse.items.forEach(item => {
          const vId       = item.snippet.resourceId.videoId;
          const uniqueKey = vId + '_' + plId;

          if (!existingVideoKeys.has(uniqueKey)) {
            const vTitle      = item.snippet.title;
            const vDesc       = item.snippet.description || '';
            const publishedAt = item.snippet.publishedAt;
            const vThumb      = item.snippet.thumbnails && item.snippet.thumbnails.high
                                ? item.snippet.thumbnails.high.url : '';

            // สกัดดึงลิงก์ Google Drive หรือเอกสารจาก Description อัตโนมัติ (ถ้ามี)
            const driveUrlMatch = vDesc.match(/https:\/\/(drive|docs)\.google\.com\/[^\s]+/g);
            const resourceLink  = driveUrlMatch ? driveUrlMatch[0] : '';

            // โครงสร้าง A-I ของ VIDEOS
            videoRows.push([
              vId,          // A: Video_ID
              plId,         // B: Playlist_ID
              vTitle,       // C: Video_Title
              vDesc,        // D: Description
              publishedAt,  // E: Published_At
              '',           // F: Duration
              vThumb,       // G: Thumbnail_URL
              resourceLink, // H: Resource_Link
              'Active'      // I: Status
            ]);
            existingVideoKeys.add(uniqueKey);
          }
        });
      }
      vNextToken = vResponse.nextPageToken;
    } while (vNextToken);
  });

  // ─── Step 4: บันทึก PLAYLISTS กลับลง Sheet (พร้อมค่า manual ที่ restore แล้ว)
  if (playlistRows.length > 0) {
    const clearRows = playlistSheet.getLastRow() > 1 ? playlistSheet.getLastRow() - 1 : 1;
    playlistSheet.getRange(2, 1, clearRows, 7).clearContent();
    playlistSheet.getRange(2, 1, playlistRows.length, 7).setValues(playlistRows);
  }

  // ─── Step 5: บันทึก VIDEOS เพิ่มต่อท้าย (ไม่เขียนทับรายการเดิม) ──────────
  if (videoRows.length > 0) {
    const lastRow = videoSheet.getLastRow();
    videoSheet.getRange(lastRow + 1, 1, videoRows.length, 9).setValues(videoRows);
  }

  Logger.log(`✅ Sync เสร็จ! Playlist ${playlistRows.length} รายการ | Video ใหม่ ${videoRows.length} รายการ`);
  Logger.log(`🔒 ค่าที่แก้มือ (Category/Target/Status) ใน PLAYLISTS ได้รับการอนุรักษ์ไว้เรียบร้อย`);
}

/**
 * 3. ฟังก์ชันดึงข้อมูลส่งให้ Web App (โหลดเร็ว: ดึงเฉพาะคลิปย้อนหลังไม่เกิน 2 ปี + Status = Active)
 */
function getWebData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const playlistSheet = ss.getSheetByName('PLAYLISTS');
  const videoSheet = ss.getSheetByName('VIDEOS');

  let playlists = [];
  let videos = [];
  let categoriesSet = new Set();
  let activePlaylistIds = new Set();

  // กำหนดขอบเขตเวลาย้อนหลัง 2 ปี นับจากวันที่ปัจจุบัน
  const now = new Date();
  const twoYearsAgo = new Date();
  twoYearsAgo.setFullYear(now.getFullYear() - 2);
  const twoYearsAgoTime = twoYearsAgo.getTime();

  // อ่านข้อมูล PLAYLISTS
  if (playlistSheet && playlistSheet.getLastRow() > 1) {
    let pData = playlistSheet.getRange(2, 1, playlistSheet.getLastRow() - 1, 7).getValues();
    pData.forEach(r => {
      let plId = r[0];
      let categoryName = (r[4] || 'อื่นๆ').toString().trim();
      let status = (r[6] || 'Active').toString().trim();

      if (status.toLowerCase() === 'active') {
        activePlaylistIds.add(plId);
        if (categoryName) categoriesSet.add(categoryName);

        playlists.push({
          id: plId,
          title: r[1],
          description: r[2],
          thumbnail: r[3],
          category: categoryName,
          target: r[5],
          status: status
        });
      }
    });
  }

  // อ่านข้อมูล VIDEOS
  if (videoSheet && videoSheet.getLastRow() > 1) {
    let lastCol = Math.max(videoSheet.getLastColumn(), 9);
    let vData = videoSheet.getRange(2, 1, videoSheet.getLastRow() - 1, lastCol).getValues();

    vData.forEach(r => {
      let vPlaylistId = r[1];
      let dateObj = r[4] ? new Date(r[4]) : new Date(0);
      let dateTime = dateObj.getTime();
      let vStatus = (r[8] || 'Active').toString().trim(); // คอลัมน์ I

      // เงื่อนไข: คลิปย้อนหลังไม่เกิน 2 ปี + Status ไม่ใช่ Hide + Playlist ต้นทางต้อง Active
      if (dateTime >= twoYearsAgoTime && vStatus.toLowerCase() !== 'hide' && activePlaylistIds.has(vPlaylistId)) {
        videos.push({
          id: r[0],
          playlistId: vPlaylistId,
          title: r[2],
          description: r[3],
          rawDate: dateTime,
          publishedAt: r[4] ? dateObj.toLocaleDateString('th-TH', { year: 'numeric', month: 'short', day: 'numeric' }) : '',
          duration: r[5],
          thumbnail: r[6],
          resourceLink: r[7]
        });
      }
    });

    // เรียงลำดับคลิปจาก ใหม่สุด -> เก่าสุด
    videos.sort((a, b) => b.rawDate - a.rawDate);
  }

  return {
    categories: Array.from(categoriesSet),
    playlists: playlists,
    videos: videos
  };
}

/**
 * ฟังก์ชันช่วยตรวจสอบ Key ซ้ำ
 */
function getExistingVideoKeys(sheet) {
  let lastRow = sheet.getLastRow();
  let set = new Set();
  if (lastRow > 1) {
    let data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
    data.forEach(row => set.add(row[0] + '_' + row[1]));
  }
  return set;
}
